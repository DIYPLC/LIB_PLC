FUNCTION_BLOCK FbTask1 (* Пользовательская задача 1. *)
VAR_INPUT
	Ts_ms: UDINT := 100; (* Шаг дискретизации по времени [мс]. *)
	Ts: REAL := 0.1; (* Шаг дискретизации по времени [с]. *)
	Reset: BOOL := TRUE; (* Сброс при перезагрузке. *)
END_VAR
VAR_OUTPUT
END_VAR
VAR
	DbBlink: FbBlink; (* Мигающий светодиод. *)
END_VAR

(*  Мигающий светодиод.
            DbBlink 
    +---------------------+
    |       FbBlink       |
   -|TimeOn      BlinkLamp|->-
   -|TimeOff              |
   -|Ts_ms                |
   -|Reset                |
    +---------------------+    *)
DbBlink.TimeOn   := 500              ; (* Время включения лампочки [мс]. *)
DbBlink.TimeOff  := 500              ; (* Время выключения лампочки [мс]. *)
DbBlink.Ts_ms    := Ts_ms            ; (* Шаг дискретизации по времени [мс]. *)
DbBlink.Reset    := Reset            ; (* Сброс при перезагрузке. *)
DbBlink()                            ;
%QX0.0           := DbBlink.BlinkLamp; (* Мигающая лампочка. *)
